    /**
 * 두 칸 혼잡도 + LED 색 카드 (subway 팀 화면)
 * 웹캠 2대가 보내는 car_1·car_2 이벤트를 받아 칸마다 사람 수·혼잡 단계를 보여 주고,
 * 자동화 규칙이 켠 LED의 지금 색도 함께 보여 줍니다. (LED는 규칙이 제어 — 여기서는 보기만)
 */
"use client";

import React, { useEffect, useState } from "react";

import { useScenario } from "@/lib/scenario";
import { asObject } from "@/lib/slots";

// ★ 우리 팀 슬롯 번호로 바꾸세요 (/kit → 하드웨어 구성)
const CARS = [
  { label: "car_1", name: "1번 칸", led: "actuator_05" },
  { label: "car_2", name: "2번 칸", led: "actuator_06" },
];
const LEVEL_NAMES = ["여유", "보통", "혼잡", "매우 혼잡"];
const STALE_MS = 10_000; // 10초 동안 소식이 없으면 "카메라 소식 없음"

interface CarReading {
  people: number;
  level: number;
  at: number;
}

export default function CarCrowdLive() {
  const { onVision, slotOf } = useScenario();
  const [cars, setCars] = useState<Record<string, CarReading>>({});
  const [now, setNow] = useState(0);

  // car_1·car_2 이벤트만 받습니다 (사람 수는 extra.people, 단계는 extra.level)
  useEffect(
    () =>
      onVision(
        CARS.map((car) => car.label),
        (event) => {
          const extra = asObject(event.extra) ?? {};
          const people = Number(extra.people ?? event.count ?? 0);
          const level = Number(extra.level ?? 0);
          const label = String(event.label ?? "");
          setCars((prev) => ({ ...prev, [label]: { people, level, at: Date.now() } }));
        }
      ),
    [onVision]
  );

  // 1초마다 지금 시각 — "소식 없음" 판정용
  useEffect(() => {
    const timer = setInterval(() => setNow(Date.now()), 1000);
    return () => clearInterval(timer);
  }, []);

  return (
    <div className="grid grid-cols-2 gap-4">
      {CARS.map((car) => {
        const reading = cars[car.label];
        const fresh = reading !== undefined && now - reading.at < STALE_MS;
        const led = slotOf(car.led);
        const ledValue = asObject(led?.current_value ?? led?.desired_value);
        const ledOn = String(led?.current_state ?? led?.desired_state ?? "") === "on";
        const ledColor = ledOn ? String(ledValue?.color ?? "#ffffff") : "transparent";
        return (
          <section key={car.label} className="rounded-xl border border-slate-200 bg-white p-5">
            <h3 className="text-lg font-semibold text-slate-800">{car.name}</h3>
            {fresh ? (
              <>
                <p className="mt-2 text-5xl font-semibold tabular-nums text-slate-900">
                  {reading.people}
                  <span className="ml-1 text-xl text-slate-500">명</span>
                </p>
                <p className="mt-1 text-lg text-slate-700">{LEVEL_NAMES[reading.level] ?? "—"}</p>
              </>
            ) : (
              <p className="mt-2 text-lg text-slate-500">카메라 소식 없음</p>
            )}
            <p className="mt-3 flex items-center gap-2 text-sm text-slate-600">
              <span
                aria-hidden="true"
                className="h-5 w-5 rounded-full border border-slate-300"
                style={{ backgroundColor: ledColor }}
              />
              LED {led ? (ledOn ? String(ledValue?.color ?? "켜짐") : "꺼짐") : "배치표에 없음"}
            </p>
          </section>
        );
      })}
    </div>
  );
}

    이미 두 LED를 수동으로 켜 봤다면 배치표는 그대로 두고 슬롯 번호만 확인합니다.

    • LED마다 데이터 핀이 따로 필요합니다 (한 줄에 이어 붙이면 키트는 칸별로 나눠 켜지 못함). 파이 5 네오픽셀 라이브러리는 서로 다른 핀 2개를 함께 씁니다.
    • 배선: DIN → 330Ω → GPIO, 5V, GND. 두 줄을 합쳐 16알이 넘거나 밝게 쓰면 별도 5V(GND는 파이와 공통).
    • 라이브러리·/dev/pio0는 12번 매뉴얼(네오픽셀 업데이트 반영) 그대로.

    한 번만 보냄

    사람 수가 그 범위에 들어올 때 한 번 색을 보냅니다. 같은 단계가 이어지면 다시 보내지 않아 LED가 깜빡이지 않습니다.

    3장 · 자동화 규칙

    3-3. /kit에서 규칙 만들기 — 8번 반복

    1. /kit → 자동화 규칙 → 규칙 추가
    2. 규칙 이름표의 이름 (예: 1번 칸 보통)
    3. 언제"카메라가 무언가를 감지하면" → 감지할 대상 칩에서 car_1 (2번 칸이면 car_2). 칩이 없으면 칸에 직접 입력
    4. 사람 수2 명 이상 4 명 미만 (매우 혼잡은 미만 칸을 비움)
    5. 무엇을액추에이터 제어 → 그 칸의 LED → 켜기 → 색을 눌러 표의 색 고르기 (#ffd600 칸에 직접 입력 가능)
    6. 규칙 만들기 → 목록에 "카메라가 'car_1'을(를) 감지하면 (2~3명) → 1번 칸 혼잡도 LED을(를) on #ffd600"

    "색" 칸은 LED가 RGB·네오픽셀이고 "켜기"일 때만 보입니다. 다음 규칙은 "규칙 추가"를 다시 누르면 사람 수·색 칸이 앞 값 그대로라 숫자만 바꾸면 됩니다.

    3장 · 자동화 규칙

    3-4. 웹캠 없이 먼저 시험 → 진짜 웹캠으로

    팀 PC · PowerShell · 저장소 맨 위 (한 창에서 차례로)
    # 준비 — 키를 화면에 치지 않도록 backend/.env에서 읽어 옴
$key = (Get-Content backend\.env | Select-String '^DEVICE_API_KEY=').Line.Split('=', 2)[1].Trim()
$url = "http://localhost:8000/api/v1/vision/events"
$h = @{ "X-Device-Api-Key" = $key }

# 1번 칸 3명 → 1번 LED 노랑
Invoke-RestMethod -Method Post -Uri $url -Headers $h -ContentType "application/json" -Body '{"event_type":"car_crowd","label":"car_1","detected":true,"count":3,"extra":{"people":3,"level":1}}'

# 2번 칸 7명 → 2번 LED 빨강
Invoke-RestMethod -Method Post -Uri $url -Headers $h -ContentType "application/json" -Body '{"event_type":"car_crowd","label":"car_2","detected":true,"count":7,"extra":{"people":7,"level":3}}'

    비전 창 2개는 꺼 두고 합니다(켜져 있으면 진짜 사람 수가 2.5초 안에 덮어씀).

    이렇게 되면 성공 (실제 로그)

    파이 데몬 창:
    [actuator_05] 1번 칸 혼잡도 LED → on RGB(255, 214, 0)
    [actuator_06] 2번 칸 혼잡도 LED → on RGB(213, 0, 0)
    백엔드 창: [규칙 발동] … '1번 칸 보통' … 'count': 3 · 2-2의 카드: LED 색 동그라미가 같은 색.

    진짜 웹캠으로

    1. 비전 창 ①·② 켜기 (6번 슬라이드)
    2. 1번 칸 웹캠 앞에 사람(또는 사람 사진)을 1명 → 2명 → 4명 → 6명으로 늘리기
    3. 단계가 바뀔 때마다 1번 LED가 초록 → 노랑 → 주황 → 빨강, 2번 LED는 그대로
    4. 2번 칸도 똑같이 — 1번 LED는 그대로여야 합니다
